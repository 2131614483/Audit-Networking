#### 2.2.2 规则引擎DSL示例

```
// 规则示例：职责分离违反检测
RULE "Segregation of Duties - Payment Create vs Approve"
WHEN
  // 同一用户在同一交易中同时担任创建者和审批者
  EXISTS transaction t
  WHERE t.created_by == t.approved_by
  AND t.type == "PAYMENT"
  AND t.amount > 10000
  AND t.status == "COMPLETED"
  AND t.create_date >= CURRENT_DATE - 7
  
THEN
  SEVERITY: HIGH
  ACTION: 
    - CREATE_ALERT("职责分离违反", 
        "用户 {t.created_by} 在同一笔付款交易(ID:{t.id})中同时担任创建者和审批者，金额{t.amount}")
    - SEND_NOTIFICATION("audit_manager", "IMMEDIATE")
    - LOG_TO_AUDIT_TRAIL(t)
END

// 规则示例：动态阈值（基于ML）
RULE "Revenue Anomaly - ML Enhanced Threshold"
WHEN
  // ML模型输出的异常分数超过动态阈值
  EXISTS business_unit bu
  WHERE bu.ml_anomaly_score > bu.dynamic_threshold
  AND bu.metric == "DAILY_REVENUE"
  AND bu.dynamic_threshold = ML_THRESHOLD(
    model: "revenue_anomaly_v2",
    confidence: 0.95,
    lookback_days: 90
  )
  
THEN
  SEVERITY: MEDIUM
  ACTION:
    - CREATE_ALERT("收入异常", 
        "BU {bu.name} 日收入异常，ML异常分数={bu.ml_anomaly_score}, 阈值={bu.dynamic_threshold}")
    - TRIGGER_INVESTIGATION(bu.id)
END
```

│  │  模型1：Isolation Forest（无监督）                            │    │
│  │  适用：无历史标签数据的异常检测                                │    │
│  │  输入：100+维特征（经PCA降维至50维）                          │    │
│  │  输出：异常分数（0-1），>0.7标记为异常                        │    │
│  │  训练：每周更新（滑动窗口90天）                                │    │
│  │  优势：无需标注数据，对高维数据有效                            │    │
│  └──────────────────────────────────────────────────────────────┘    │
